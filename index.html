async function sendMessage() { const input = document.getElementById('userInput'); const btn = document.getElementById('sendBtn'); const text = input.value.trim(); if (!text) return; appendMessage(text, 'user-msg'); input.value = ''; btn.disabled = true; const loadingDiv = appendMessage('回答を生成中...', 'agent-msg'); try { // モデル名を gemini-1.5-flash に固定し、URLを整形 const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${API_KEY}`; const response = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: [{ text: `${SYSTEM_INSTRUCTION}\n\nユーザーからの質問: ${text}` }] }] }) }); const data = await response.json(); if (data.error) { loadingDiv.textContent = `エラー: ${data.error.message || 'APIリクエストに失敗しました'}`; console.error('API Error:', data.error); } else { const reply = data.candidates?.[0]?.content?.parts?.[0]?.text || '回答を取得できませんでした。'; loadingDiv.textContent = reply; } } catch (err) { loadingDiv.textContent = '通信エラーが発生しました。コンソール(F12)で詳細を確認してください。'; console.error('Fetch Error:', err); } finally { btn.disabled = false; } }
